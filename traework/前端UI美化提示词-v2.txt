  BG = #000000          # 全局背景（纯黑）
  CARD = #1C1C1E        # 卡片背景
  SURFACE = #2C2C2E     # 按钮/输入框背景

  TEXT2 = #8E8E93       # 次级文字
  TEXT3 = #636366       # 三级文字
  BORDER = #38383A      # 分隔线/边框
  GREEN = #34C759       # 安全/成功
  ORANGE = #FF9500      # 警告
  RED = #FF3B30         # 危险/错误
  BLUE = #0A84FF        # 强调/确认

字体：
  全局：Inter, -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif
  大数字：font-weight: 300, font-size: 64pt
  卡片标题：font-weight: 600, font-size: 17px, color: #FFFFFF
  正文：font-weight: 400, font-size: 15px
  标签：font-weight: 500, font-size: 13px, color: #8E8E93, uppercase, letter-spacing: 0.5px

圆角：
  大卡片：24px
  按钮/输入框/下拉框：12px
  小标签/pill：100px（完全圆角）

布局原则：
  - 取消所有 QGroupBox 的边框和标题样式（改用 QWidget + 卡片背景）
  - 用背景色差异代替边框区分层次
  - 卡片内边距：20px
  - 卡片间距：16px
  - 禁止 text-shadow、禁止渐变背景（除品牌标题外）
